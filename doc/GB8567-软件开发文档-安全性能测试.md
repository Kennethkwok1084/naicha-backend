#### 7.1.1 JWT认证机制

**令牌结构**：
```json

    "alg": "HS256",
    "typ": "JWT"
  },
  "payload": {
    "sub": "user_id" | "admin_id",
    "type": "user" | "admin",
```

**令牌生成**：
```python
from jose import jwt
from datetime import datetime, timedelta

def create_access_token(
    subject: str, 
    token_type: str,
    role: str = None
) -> str:
    expire = datetime.utcnow() + timedelta(
        minutes=settings.jwt_expire_minutes
    )
    payload = {
        "sub": subject,
        "type": token_type,
        "role": role,
        "exp": expire,
        "iat": datetime.utcnow()
    }
    return jwt.encode(
        payload, 
        settings.secret_key, 
        algorithm="HS256"
    )
```

**令牌验证**：
```python
async def get_current_user(
    token: str = Depends(oauth2_scheme)
) -> User:
    try:
        payload = jwt.decode(
            token, 
            settings.secret_key, 
            algorithms=["HS256"]
        )
        if payload.get("type") != "user":
            raise credentials_exception
        user_id = payload.get("sub")
        # 从数据库加载用户
        user = await user_service.get_by_id(user_id)
        if not user:
            raise credentials_exception
        return user
    except JWTError:
        raise credentials_exception
```

#### 7.1.2 基于角色的访问控制（RBAC）

**角色定义**：

| 角色 | 权限范围 |
|------|----------|
| **admin** | 系统管理员，拥有所有权限 |
| **manager** | 店长，拥有订单、库存、数据看板权限，无系统配置权限 |
| **clerk** | 店员，仅POS建单（限现金/刷卡）、查看订单权限 |

**权限装饰器**：
```python
from functools import wraps

def require_role(*allowed_roles: str):
    def decorator(func):
        @wraps(func)
        async def wrapper(
            current_admin: Admin = Depends(get_current_admin),
            *args,
            **kwargs
        ):
            if current_admin.role not in allowed_roles:
                raise HTTPException(
                    status_code=403,
                    detail="Insufficient permissions"
                )
            return await func(
                current_admin=current_admin,
                *args,
                **kwargs
            )
        return wrapper
    return decorator

# 使用示例
@router.post("/admin/orders")
@require_role("admin", "manager", "clerk")
async def create_pos_order(...):
    pass
```

#### 7.3.2 幂等性保护

**幂等键机制**：
```python
async def ensure_idempotency(
    key: str,
    scope: str,
    request_body: dict,
    response_builder: Callable
) -> dict:
    # 检查幂等键是否存在
    existing = await idem_service.get(key, scope)
    if existing:
        # 验证请求体是否一致
        if existing.request_hash != hash_dict(request_body):
            raise HTTPException(
                status_code=409,
                detail="Idempotency key conflict"
            )
        # 返回缓存的响应
        return existing.response_snapshot
    
    # 执行业务逻辑
    response = await response_builder()
    
    # 保存幂等记录
    await idem_service.save(
        key=key,
        scope=scope,
        request_hash=hash_dict(request_body),
        response_snapshot=response,
        expire_at=datetime.utcnow() + timedelta(hours=24)
    )
    
    return response
```

**幂等键要求**：
- 创建类接口必须提供（HTTP Header: `Idempotency-Key`）
- 格式：UUID v4
- 有效期：24小时
- 存储：PostgreSQL `idempotency_keys`表

#### 8.1.1 菜单缓存

**实现方式**：应用内存缓存 + Redis版本协作

```python
from functools import lru_cache
from datetime import datetime

_MENU_CACHE: dict = {}
_MENU_VERSION_KEY = "menu:version"

async def get_menu(settings: Settings) -> dict:
    """获取菜单（带缓存）"""
    cache_key = "menu_data"
    now = datetime.utcnow().timestamp()
    
    # 检查内存缓存
    if cache_key in _MENU_CACHE:
        cached_at, remote_version, data = _MENU_CACHE[cache_key]
        
        # 检查Redis版本号
        current_version = await _read_remote_menu_version(settings)
        if current_version and current_version != remote_version:
            # 版本失配，清除缓存
            del _MENU_CACHE[cache_key]
        elif now - cached_at < settings.menu_cache_ttl_seconds:
            # TTL内有效
            return data
    
    # 重新加载菜单
    data = await _load_menu_from_db()
    remote_version = await _read_remote_menu_version(settings)
    _MENU_CACHE[cache_key] = (now, remote_version, data)
    
    return data

async def invalidate_menu_cache(settings: Settings):
    """失效菜单缓存（库存变更时调用）"""
    global _MENU_CACHE
    _MENU_CACHE.clear()
    await _bump_remote_menu_version(settings)
```

**缓存参数**：
- TTL: 240秒（4分钟）
- 触发刷新：库存状态变更、商品上下架
- 版本控制：Redis `menu:version` 键

```python
from redis import Redis
from redis.lock import Lock

def _acquire_task_lock(
    name: str, 
    interval_seconds: int
) -> tuple[bool, Lock | None]:
    client = Redis.from_url(settings.celery_broker_url)
    ttl = max(interval_seconds * 2, 120)
    
    try:
        lock = client.lock(name, timeout=ttl, blocking=False)
        acquired = lock.acquire(blocking=False)
        if not acquired:
            return False, None
        return True, lock
    except RedisError:
        # Redis故障时允许执行（降级）
        return True, None

@celery_app.task
def print_job_recovery():
    acquired, lock = _acquire_task_lock(
        "celery:lock:print_job_recovery",
        interval_seconds=60
    )
    if not acquired:
        logger.info("print_job_recovery skipped (locked)")
        return
    
    try:
        # 执行任务...
        pass
    finally:
        if lock and lock.owned():
            lock.release()
```

#### 8.4.2 WebSocket推送

**连接管理**：
```python
from fastapi import WebSocket

class MerchantNotifier:
    def __init__(self):
        self.connections: dict[int, WebSocket] = {}
    
    async def connect(self, admin_id: int, ws: WebSocket):
        await ws.accept()
        self.connections[admin_id] = ws
    
    async def disconnect(self, admin_id: int):
        self.connections.pop(admin_id, None)
    
    async def broadcast(self, message: dict):
        """广播消息到所有在线管理员"""
        for admin_id, ws in list(self.connections.items()):
            try:
                await ws.send_json(message)
            except Exception:
                await self.disconnect(admin_id)
```

```python
import redis.asyncio as redis

class MerchantNotifier:
    async def startup(self):
        self.redis = await redis.from_url(
            settings.ws_broadcast_url
        )
        self.pubsub = self.redis.pubsub()
        await self.pubsub.subscribe(
            settings.ws_broadcast_channel
        )
        asyncio.create_task(self._listen_broadcast())
    
    async def _listen_broadcast(self):
        async for message in self.pubsub.listen():
            if message["type"] == "message":
                data = json.loads(message["data"])
                await self.broadcast(data)
    
    async def publish(self, message: dict):
        """发布消息到Redis（多实例共享）"""
        await self.redis.publish(
            settings.ws_broadcast_channel,
```

```python
async def test_concurrent_order_creation(db_session_factory):
    """测试并发下单不会超卖"""

    initial_stock = 10

    tasks = [

        for _ in range(20)
    ]
    
    results = await asyncio.gather(*tasks, return_exceptions=True)
    
    # 统计成功订单数
    success_count = sum(

    # 应该只有10个订单成功（不超卖）
    assert success_count == initial_stock
```

```python
from locust import HttpUser, task, between

class PaymentCallbackUser(HttpUser):
    wait_time = between(0.1, 0.5)
    
    def on_start(self):
        """创建订单并获取订单号"""
        response = self.client.post(
            "/api/v1/orders",
            headers={
                "Authorization": f"Bearer {self.get_user_token()}",
                "Idempotency-Key": f"load-test-{uuid.uuid4()}"
            },
            json={
                "items": [{"product_id": 101, "quantity": 1}],
                "order_type": "pickup"
            }
        )
        self.order_number = response.json()["order_number"]
    
    @task
    def payment_callback(self):
        """模拟支付回调"""
        payload = {
            "event_id": f"evt_{uuid.uuid4()}",
            "order_number": self.order_number,
            "transaction_id": f"wx_txn_{uuid.uuid4()}",
            "amount": "13.00",
            "currency": "CNY",
            "channel": "wechat_jsapi",
            "status": "SUCCESS",
            "paid_at": datetime.utcnow().isoformat()
        }
        
        signature = generate_hmac_signature(payload)
        
        self.client.post(
            "/api/v1/payments/notify/wechat",
            headers={"X-Wechat-Signature": signature},
            json=payload
        )
```
